0019 | def wilson(successes: int, n: int, z: float = 1.96):
     # Wilson score interval -- behaves at the small cell counts in the tail.
0020 |     """Wilson score interval -- behaves at the small cell counts in the tail."""
0021 |     if n == 0:
     # Checks this condition before running the next indented block.
0022 |         return (float("nan"),) * 3
     # Hands this result back to the stage that called this function.
0023 |     p = successes / n
     # Keeps this intermediate result under the name `p`.
0024 |     den = 1 + z * z / n
     # Keeps this intermediate result under the name `den`.
0025 |     centre = (p + z * z / (2 * n)) / den
     # Keeps this intermediate result under the name `centre`.
0026 |     half = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / den
     # Keeps this intermediate result under the name `half`.
0027 |     return p, max(0.0, centre - half), min(1.0, centre + half)
     # Hands this result back to the stage that called this function.
