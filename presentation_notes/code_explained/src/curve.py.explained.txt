0019 | def wilson(successes: int, n: int, z: float = 1.96):  # Wilson score interval -- behaves at the small cell counts in the tail.
0020 |     """Wilson score interval -- behaves at the small cell counts in the tail."""
0021 |     if n == 0:  # Runs the next indented steps only when this condition is true.
0022 |         return (float("nan"),) * 3  # Gives this answer back to the code that called the function.
0023 |     p = successes / n  # Saves a value named p so the next lines can use it.
0024 |     den = 1 + z * z / n  # Saves a value named den so the next lines can use it.
0025 |     centre = (p + z * z / (2 * n)) / den  # Saves a value named centre so the next lines can use it.
0026 |     half = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / den  # Saves a value named half so the next lines can use it.
0027 |     return p, max(0.0, centre - half), min(1.0, centre + half)  # Gives this answer back to the code that called the function.
