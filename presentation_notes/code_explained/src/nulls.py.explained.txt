0020 | def shuffle_timestamps(df: pd.DataFrame, rng: np.random.Generator) -> pd.DataFrame:  # Keep each game's players/opening, but remove the original time order.
0021 |     """Keep each game's players/opening, but remove the original time order.
0022 |
0023 |     Say: if the observed rise also appears after this shuffle, activity and
0024 |     opening popularity may explain it without opponent-to-opponent spread.
0025 |     """
0026 |     out = df.copy()  # Stores the output value for the next step.
0027 |     out["ts"] = rng.permutation(out["ts"].to_numpy())  # Mixes up game dates but keeps the players and opening in each game.

0031 | def rewire_preserving_degree(df: pd.DataFrame, rng: np.random.Generator,  # Degree-preserving double-edge swap on the game list, timestamps kept.
0032 |                              n_swaps_per_edge: int = 5) -> pd.DataFrame:  # Saves a value named n_swaps_per_edge so the next lines can use it.
0033 |     """Degree-preserving double-edge swap on the game list, timestamps kept."""
0034 |     w = df["white"].to_numpy().copy()  # Saves a value named w so the next lines can use it.
0035 |     b = df["black"].to_numpy().copy()  # Saves a value named b so the next lines can use it.
0036 |     m = len(w)  # Saves a value named m so the next lines can use it.
0037 |     for _ in range(n_swaps_per_edge * m):  # Repeats the next indented steps once for each item.
0038 |         i, j = rng.integers(0, m, 2)  # Runs this instruction as part of the current step.
0039 |         if i == j:  # Runs the next indented steps only when this condition is true.
0040 |             continue  # Skips this item and moves to the next loop iteration.
0041 |         if b[i] == w[j] or b[j] == w[i]:  # Runs the next indented steps only when this condition is true.
0042 |             continue  # Skips this item and moves to the next loop iteration.
0043 |         b[i], b[j] = b[j], b[i]  # Stores this value under its player, opening, or field key.
0044 |     out = df.copy()  # Stores the output value for the next step.
0045 |     out["white"], out["black"] = w, b  # Stores this value under its player, opening, or field key.
0046 |     return out  # Gives this answer back to the code that called the function.

0049 | def run_null(df: pd.DataFrame, n_reps: int = None, kind: str = "shuffle",  # Build the null trials with EXACTLY the same parameters as the real ones.
0050 |              seed: int = config.RANDOM_SEED,  # Saves a value named seed so the next lines can use it.
0051 |              window_days: int = None, by: str = "d",  # Saves a value named window_days so the next lines can use it.
0052 |              granularity: str = None, should_stop=None,  # Saves a value named granularity so the next lines can use it.
0053 |              on_progress=None) -> pd.DataFrame:  # Saves a value named on_progress so the next lines can use it.
0054 |     """Build the null trials with EXACTLY the same parameters as the real ones.
0055 |
0056 |     Both window_days and granularity must match, or the comparison is not a
0057 |     comparison. Two ways this has already gone wrong:
0058 |
0059 |     * a window wider than the observed span censors every non-adopter, so the
0060 |       null adoption rate goes to 1.0 and the lift collapses;
0061 |     * leaving granularity at the default while the real trials are keyed by
0062 |       ply compares a 0.6%-adoption ply world against a ECO-adoption
0063 |       world, which manufactures a spurious "complex contagion" verdict.
0064 |     """
0065 |     # Simple explanation: repeat the analysis after mixing up game dates.
0066 |     # If the same rise appears, the rise alone is weak evidence of spreading.
0067 |     n_reps = n_reps or config.N_NULL_REPLICATES  # Saves a value named n_reps so the next lines can use it.
0068 |     rng = np.random.default_rng(seed)  # Saves a value named rng so the next lines can use it.
0069 |     frames = []  # Saves a value named frames so the next lines can use it.
0070 |     for rep in range(n_reps):  # Repeats the next indented steps once for each item.
0071 |         if should_stop is not None and should_stop():  # Runs the next indented steps only when this condition is true.
0072 |             break  # Ends the current loop.
0073 |         null_df = (shuffle_timestamps(df, rng) if kind == "shuffle"  # Creates one shuffled-date version of the same games.
0074 |                    else rewire_preserving_degree(df, rng))  # Runs this instruction as part of the current step.
0075 |         trials = events.build_trials(null_df, window_days=window_days,  # Stores the exposure-trial table for the next step.
0076 |                                      granularity=granularity)  # Saves a value named granularity so the next lines can use it.
0077 |         if not len(trials):  # Checks whether enough records exist for a trustworthy next step.
0078 |             continue  # Skips this item and moves to the next loop iteration.
0079 |         tb = curve.analyse(trials)["by_" + by]  # Recalculates the adoption curve from these shuffled-date trials.
0080 |         tb["rep"] = rep  # Stores this value under its player, opening, or field key.
0081 |         frames.append(tb)  # Adds this item to the list being built.
0082 |         if on_progress is not None:  # Runs the next indented steps only when this condition is true.
0083 |             on_progress(rep + 1, n_reps)  # Runs this instruction as part of the current step.
0084 |     if not frames:  # Runs the next indented steps only when this condition is true.
0085 |         return pd.DataFrame()  # Creates a labelled table from the collected rows.
0086 |     allreps = pd.concat(frames, ignore_index=True)  # Saves a value named allreps so the next lines can use it.
0087 |     return (allreps.groupby(by)  # Groups rows sharing these keys before counting or averaging them.
0088 |             .agg(n=("n", "sum"), rate=("rate", "mean"), rate_sd=("rate", "std"))  # Runs this instruction as part of the current step.
0089 |             .reset_index())  # Runs this instruction as part of the current step.

0092 | def compare(real_by_d: pd.DataFrame, null_by_d: pd.DataFrame,  # Put observed and shuffled rates side by side for each d value.
0093 |             by: str = "d") -> pd.DataFrame:  # Saves a value named by so the next lines can use it.
0094 |     """Put observed and shuffled rates side by side for each d value.
0095 |
0096 |     `lift_over_null` is observed rate / shuffled rate. Above 1 means excess
0097 |     over that reference, not a causal effect estimate.
0098 |     """
0099 |     # Each row now shows two rates for the same number of different opponents:
0100 |     # what really happened and what happened after dates were shuffled.
0101 |     m = real_by_d.merge(null_by_d, on=by, suffixes=("_real", "_null"))  # Joins two tables on shared identifiers so their columns can be compared.
0102 |     m["lift_over_null"] = m["rate_real"] / m["rate_null"].replace(0, np.nan)  # Compares the observed adoption rate with the shuffled-time reference rate.
0103 |
0104 |     # Spread of the null rate across replicates. With a single replicate there
0105 |     # is no spread to measure (std of one value is NaN), which would silently
0106 |     # make every z NaN and collapse the verdict to "nothing detected". Fall
0107 |     # back to the binomial standard error of the null rate, which is a valid
0108 |     # estimate of the same quantity and lets one replicate still be informative.
0109 |     se = m["rate_sd"].replace(0, np.nan)  # Saves a value named se so the next lines can use it.
0110 |     binom = np.sqrt(m["rate_null"] * (1 - m["rate_null"]) / m["n_null"].clip(lower=1))  # Caps large counts into the final chart bucket, such as six or more opponents.
0111 |     m["null_se"] = se.fillna(binom)  # Stores this value under its player, opening, or field key.
0112 |     m["z_vs_null"] = (m["rate_real"] - m["rate_null"]) / m["null_se"].replace(0, np.nan)  # Stores this value under its player, opening, or field key.
0113 |     return m[[by, "n_real", "rate_real", "rate_null", "rate_sd",  # Gives this answer back to the code that called the function.
0114 |               "lift_over_null", "z_vs_null"]]    // Compares the observed adoption rate with the shuffled-time reference rate.
