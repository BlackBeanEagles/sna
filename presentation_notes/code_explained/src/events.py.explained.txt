0001 | """Steps 3-4 -- exposure/adoption logs and the trial table.
0002 |
0003 | A behaviour is (opening, colour). One game between white=u and black=v with
0004 | opening X produces two performances and two exposures:
0005 |
0006 |     u performs (X, W)  ->  exposes v to (X, W)
0007 |     v performs (X, B)  ->  exposes u to (X, B)
0008 |
0009 | Adoption of behaviour b by player p = the FIRST time p performs b.
0010 |
0011 | The trial table is the unit of analysis. One row per (player, behaviour,
0012 | k-th exposure) where the player had not yet performed the behaviour:
0013 |
0014 |     k   cumulative exposure events so far
0015 |     d   distinct exposing opponents so far
0016 |     sd  structural diversity: connected components of the subgraph induced
0017 |         on those d exposers, using only edges that exist by time t_k
0018 |         (Ugander et al. 2012)
0019 |     adopted  1 if the player first performs b within ADOPTION_WINDOW_DAYS
0020 |
0021 | Holding k fixed and varying d (or sd) is the decisive comparison.
0022 | """

0070 | def _components(nodes: list[str], edge_first_t: dict, t_now) -> int:  # Connected components of the subgraph induced on nodes, counting only edges that already existed at t_now.
0071 |     """Connected components of the subgraph induced on `nodes`, counting only
0072 |     edges that already existed at t_now. O(d^2) -- d is small by construction."""
0073 |     n = len(nodes)  # Saves a value named n so the next lines can use it.
0074 |     if n <= 1:  # Runs the next indented steps only when this condition is true.
0075 |         return n  # Gives this answer back to the code that called the function.
0076 |     parent = list(range(n))  # Stores the union-find parent list for grouping connected exposers for the next step.
0077 |
0078 |     def find(i):  # Defines a reusable step; the following indented lines are its body.
0079 |         while parent[i] != i:  # Repeats the indented block until this condition stops holding.
0080 |             parent[i] = parent[parent[i]]  # Shortens the search path while finding a connected exposer group.
0081 |             i = parent[i]  # Saves a value named i so the next lines can use it.
0082 |         return i  # Gives this answer back to the code that called the function.
0083 |
0084 |     for i in range(n):  # Repeats the next indented steps once for each item.
0085 |         for j in range(i + 1, n):  # Repeats the next indented steps once for each item.
0086 |             t = edge_first_t.get(frozenset((nodes[i], nodes[j])))  # Saves a temporary value used by the next calculation.
0087 |             if t is not None and t <= t_now:  # Runs the next indented steps only when this condition is true.
0088 |                 ri, rj = find(i), find(j)  # Runs this instruction as part of the current step.
0089 |                 if ri != rj:  # Runs the next indented steps only when this condition is true.
0090 |                     parent[ri] = rj  # Joins two exposer groups after finding a link between them.
0091 |     return len({find(i) for i in range(n)})  # Counts how many separate social sources the exposing opponents form.
