0001 | """Features for the held-out prediction check, built without leakage.
0002 |
0003 | Two problems with the original check, both fixed here.
0004 |
0005 | **Leakage.** The original model computed summary features over the whole
0006 | month. Here fixed background features are computed before the first evaluated
0007 | trial; exposure features are measured at each trial. Whole player-behaviour
0008 | spells are assigned to one side of the held-out split.
0009 |
0010 | **A missing explanation.** The original feature set asked only "who played this
0011 | opening at you?". A player may choose a new opening simply because it is close
0012 | to openings they already know, without copying an opponent. Leaving that out
0013 | makes social features look better than they are. `repertoire_density` measures
0014 | this competing explanation. It is a project control, not a causal proof.
0015 |
0016 | Three feature blocks are produced, so the three can be compared directly:
0017 |
0018 |     BASELINE    popularity, rating, activity        "popular openings get played"
0019 |     REPERTOIRE  + density, repertoire size          "you learn what is near what you know"
0020 |     NETWORK     + k, d, sd, degree, kcore, between  "who played it at you"
0021 | """

0043 | def relatedness(rep: dict[str, set], openings: list[str]) -> pd.DataFrame:  # Opening-to-opening proximity from co-play, following De Marzo & Servedio.
0044 |     """Opening-to-opening proximity from co-play, following De Marzo & Servedio.
0045 |
0046 |     Two openings are related when the same players tend to play both. Using
0047 |     min() in the denominator keeps a rare opening that is always played
0048 |     alongside a common one from being scored as unrelated to it.
0049 |     """
0050 |     idx = {o: i for i, o in enumerate(openings)}  # Saves a value named idx so the next lines can use it.
0051 |     n = len(openings)  # Saves a value named n so the next lines can use it.
0052 |     co = np.zeros((n, n), dtype=float)  # Saves a value named co so the next lines can use it.
0053 |     count = np.zeros(n, dtype=float)  # Saves a value named count so the next lines can use it.
0054 |
0055 |     for played in rep.values():  # Repeats the next indented steps once for each item.
0056 |         ids = [idx[o] for o in played if o in idx]  # Saves a value named ids so the next lines can use it.
0057 |         for i in ids:  # Repeats the next indented steps once for each item.
0058 |             count[i] += 1  # Adds this amount to the running total.
0059 |         for a in range(len(ids)):  # Repeats the next indented steps once for each item.
0060 |             for b in range(a + 1, len(ids)):  # Repeats the next indented steps once for each item.
0061 |                 co[ids[a], ids[b]] += 1  # Adds this amount to the running total.
0062 |                 co[ids[b], ids[a]] += 1  # Adds this amount to the running total.
0063 |
0064 |     denom = np.minimum.outer(count, count)  # Saves a value named denom so the next lines can use it.
0065 |     with np.errstate(divide="ignore", invalid="ignore"):  # Opens a resource and guarantees cleanup when the block ends.
0066 |         phi = np.where(denom > 0, co / denom, 0.0)  # Saves a value named phi so the next lines can use it.
0067 |     np.fill_diagonal(phi, 0.0)  # Runs this instruction as part of the current step.
0068 |     return pd.DataFrame(phi, index=openings, columns=openings)  # Creates a labelled table from the collected rows.

0071 | def density_matrix(rep: dict[str, set], phi: pd.DataFrame,  # Density for every (opening, player) pair at once.
0072 |                    players: list[str]) -> np.ndarray:  # Runs this instruction as part of the current step.
0073 |     """Density for every (opening, player) pair at once.
0074 |
0075 |     How close an opening sits to what a player already plays: the share of that
0076 |     opening's total relatedness pointing at openings already in the repertoire.
0077 |     1.0 means everything related to it is already known, 0.0 means none of it.
0078 |
0079 |     Done as a single matrix product. The obvious per-trial version costs a
0080 |     pandas label lookup and a fancy-index sum for each of ~224k trials, which
0081 |     took over ten minutes; this takes a fraction of a second and is the same
0082 |     arithmetic. phi has a zero diagonal, so an opening never counts itself.
0083 |     """
0084 |     openings = list(phi.index)  # Saves a value named openings so the next lines can use it.
0085 |     o_idx = {o: i for i, o in enumerate(openings)}  # Saves a value named o_idx so the next lines can use it.
0086 |     mask = np.zeros((len(players), len(openings)), dtype=float)  # Saves a value named mask so the next lines can use it.
0087 |     for pi, p in enumerate(players):  # Repeats the next indented steps once for each item.
0088 |         for o in rep.get(p, ()):  # Repeats the next indented steps once for each item.
0089 |             j = o_idx.get(o)  # Saves a value named j so the next lines can use it.
0090 |             if j is not None:  # Runs the next indented steps only when this condition is true.
0091 |                 mask[pi, j] = 1.0  # Stores this value under its player, opening, or field key.
0092 |
0093 |     phi_v = phi.to_numpy()  # Saves a value named phi_v so the next lines can use it.
0094 |     totals = phi_v.sum(axis=1)                 # per opening
0095 |     reach = phi_v @ mask.T                     # openings x players
0096 |     with np.errstate(divide="ignore", invalid="ignore"):  # Opens a resource and guarantees cleanup when the block ends.
0097 |         return np.where(totals[:, None] > 0, reach / totals[:, None], 0.0)  # Gives this answer back to the code that called the function.
