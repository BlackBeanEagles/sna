0019 | def build_graph(df: pd.DataFrame) -> nx.Graph:  # Make one player node per account and one edge per opponent pair.
0020 |     """Make one player node per account and one edge per opponent pair.
0021 |
0022 |     Repeated games increase the edge's `games` weight; they do not make extra
0023 |     nodes or edges. This is the network displayed during the live demo.
0024 |     """
0025 |     g = nx.Graph()  # Stores the player graph for the next step.
0026 |     w = df.groupby(["white", "black"]).size().reset_index(name="games")  # Groups rows sharing these keys before counting or averaging them.
0027 |     for u, v, n in w.itertuples(index=False):  # Repeats the next indented steps once for each item.
0028 |         if g.has_edge(u, v):  # Runs the next indented steps only when this condition is true.
0029 |             g[u][v]["games"] += int(n)  # Adds this amount to the running total.
0030 |         else:  # Handles the alternative when the preceding condition was false.
0031 |             g.add_edge(u, v, games=int(n))  # Runs this instruction as part of the current step.

0035 | def node_features(g: nx.Graph, betweenness_k: int = 200) -> pd.DataFrame:  # Describe each player's position and detect groups in the network.
0036 |     """Describe each player's position and detect groups in the network.
0037 |
0038 |     Degree counts opponents, k-core measures membership in a dense centre,
0039 |     and betweenness approximates how often a player bridges shortest paths.
0040 |     Louvain groups players who are more connected to one another.
0041 |     """
0042 |     deg = dict(g.degree())  # Saves a value named deg so the next lines can use it.
0043 |     core = nx.core_number(g)  # Measures how deeply each player lies in a mutually connected core.
0044 |     k = min(betweenness_k, g.number_of_nodes())  # Stores the total exposure count for the next step.
0045 |     btw = nx.betweenness_centrality(g, k=k, seed=config.RANDOM_SEED) if k > 2 else {}  # Estimates how often players bridge shortest routes through the graph.
0046 |     clus = nx.clustering(g)  # Saves a value named clus so the next lines can use it.
0047 |     comms = nx.community.louvain_communities(g, seed=config.RANDOM_SEED)  # Finds groups of players with relatively dense internal connections.
0048 |     comm_of = {n: i for i, c in enumerate(comms) for n in c}  # Saves a value named comm_of so the next lines can use it.
0049 |     feats = pd.DataFrame({  # Creates a labelled table from the collected rows.
0050 |         "player": list(g.nodes()),    // Part of the value, collection, or call spanning several lines.
0051 |         "degree": [deg.get(n, 0) for n in g.nodes()],    // Part of the value, collection, or call spanning several lines.
0052 |         "kcore": [core.get(n, 0) for n in g.nodes()],    // Part of the value, collection, or call spanning several lines.
0053 |         "betweenness": [btw.get(n, 0.0) for n in g.nodes()],    // Part of the value, collection, or call spanning several lines.
0054 |         "clustering": [clus.get(n, 0.0) for n in g.nodes()],    // Part of the value, collection, or call spanning several lines.
0055 |         "community": [comm_of.get(n, -1) for n in g.nodes()],    // Part of the value, collection, or call spanning several lines.
0056 |     })  # Runs this instruction as part of the current step.
0057 |     feats.attrs["n_communities"] = len(comms)  # Stores this value under its player, opening, or field key.
